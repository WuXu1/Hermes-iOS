# Hermes iOS on the VM with five agents: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Hermes iOS app, relay and connector work with the owner's self-hosted Hermes VM (five agents as Hermes profiles), reachable over HTTPS through Caddy.

**Architecture:** The relay and connector keep their WebSocket RPC protocol and both run on the VM: Caddy (public :443) → relay (127.0.0.1:8000) ← connector (dials the relay on loopback) → Hermes dashboard API (127.0.0.1:9119) and `hermes -p <profile> chat`. Every chat, history, memory and skills call carries a `profile`. The app gains an agent picker, Hermes-backed history and voice notes, and loses live voice, CarPlay and the Railway screens.

**Tech Stack:** Python 3.12 (FastAPI, SQLAlchemy, httpx, websockets, ruamel.yaml, pytest), Swift 6 / SwiftUI for iOS 26 (Xcode project, XCTest), systemd, Caddy 2, Azure CLI.

**Spec:** `docs/superpowers/specs/2026-10-02-vm-revamp-design.md`

## Global Constraints

- Hermes on the VM: v0.21.5 (2026.9.24, commit f97608f), user `hermes`, `HERMES_HOME=/data/.hermes`, CLI `/home/hermes/.local/bin/hermes`, source `/opt/hermes/hermes-agent`. Never run `hermes update`, and never change Hermes's own venv.
- Profiles: `default` (shown as "Commander"), `developer`, `tutor`, `assistant`, `coach`. Profile names match `^[a-z0-9][a-z0-9_-]{0,63}$`.
- The repo is public: never commit IPs, hostnames, tokens, Telegram IDs, or the owner's name, university or modules. VM-specific values are passed as script arguments or live in `/etc/hermes-mobile/*.env` on the VM.
- Phone-context MCP goes into exactly `default`, `assistant` and `coach`.
- Personas are read-only from the app. No `DELETE` of sessions, kanban tasks or cron jobs from the app.
- Live voice mode and CarPlay are removed from the app. Relay and connector talk code stays dormant, and Caddy blocks `/v1/talk/*`.
- Voice notes use Hermes's own `POST /api/audio/transcribe?profile=<p>` (local faster-whisper). Gemini is never used.
- Audio uploads are limited to 5 MB.
- New Swift files are added with `scripts/xcode_add_files.py`. Removed files go through its new `--remove` flag (Task 15). Never regenerate with XcodeGen.
- Python tests run with `uv run --extra dev pytest -q` inside `connector/` or `relay/`.
- App tests run with `xcodebuild -project HermesMobile.xcodeproj -scheme HermesMobile -destination 'platform=iOS Simulator,name=iPhone 17 Pro' test`.
- **Nothing on the VM or in Azure changes before Task 17, and each VM task starts by asking the owner for a go-ahead in chat.**

```python
# connector/src/hermes_mobile_connector/profiles.py
"""Hermes profiles (one per agent): names, homes and locked skills."""

from __future__ import annotations

import re
from pathlib import Path

from ruamel.yaml import YAML

DEFAULT_PROFILE = "default"
_NAME = re.compile(r"^[a-z0-9][a-z0-9_-]{0,63}$")


def normalize_profile(name: str | None) -> str:
    value = (name or "").strip()
    if not value:
        return DEFAULT_PROFILE
    if not _NAME.match(value):
        raise ValueError(f"invalid: {value!r} is not a profile name")
    return value


def profile_home(hermes_home: Path, profile: str | None) -> Path:
    name = normalize_profile(profile)
    if name == DEFAULT_PROFILE:
        return hermes_home
    home = hermes_home / "profiles" / name
    if not home.is_dir():
        raise ValueError(f"not_found: profile {name!r} does not exist")
    return home


def locked_skills(home: Path) -> frozenset[str]:
    config_path = home / "config.yaml"
    if not config_path.exists():
        return frozenset()
    loaded = YAML(typ="safe").load(config_path.read_text(encoding="utf-8")) or {}
    auto_load = (loaded.get("skills") or {}).get("auto_load") or []
    return frozenset(str(item) for item in auto_load if item)
```

```python
_BODY_PROFILE_ROUTES = (re.compile(rf"^/api/sessions/{_SEG}$"), re.compile(r"^/api/skills/toggle$"))
_SESSION_PATCH_KEYS = frozenset({"title", "archived"})
_SESSION_ROUTE = re.compile(rf"^/api/sessions/{_SEG}$")


def _match(method: str, path: str) -> frozenset[str]:
    method = (method or "").upper()
    for methods, pattern, keys in _ROUTES:
        if method in methods and pattern.match(path or ""):
            return keys
    raise HermesApiError(f"forbidden: {method} {path} is not available to the app")


def check_route(method: str, path: str) -> None:
    _match(method, path)


def filter_query(path: str, query: Mapping[str, Any] | None, *, method: str = "GET") -> dict[str, str]:
    keys = _match(method, path)
    return {key: str(value) for key, value in (query or {}).items() if key in keys and value is not None}


def prepare_request(params: Mapping[str, Any], *, hermes_home: Path) -> dict:
    method = str(params.get("method") or "GET").upper()
    path = str(params.get("path") or "")
    query = filter_query(path, params.get("query"), method=method)
    try:
        profile = normalize_profile(query.get("profile"))
        home = profile_home(hermes_home, profile)
    except ValueError as error:
        raise HermesApiError(str(error)) from error
    # /api/cron/jobs without a profile lists every agent's jobs ("all"); keep that.
    if "profile" in query or not (method == "GET" and path == "/api/cron/jobs"):
        query["profile"] = profile

    body = params.get("body")
    if method == "PATCH" and _SESSION_ROUTE.match(path):
        body = {key: value for key, value in dict(body or {}).items() if key in _SESSION_PATCH_KEYS}
    if any(pattern.match(path) for pattern in _BODY_PROFILE_ROUTES) and method in {"PATCH", "PUT"}:
        body = {**dict(body or {}), "profile": profile}
    if method == "PUT" and path == "/api/skills/toggle":

        if not (body or {}).get("enabled", True) and name in locked_skills(home):
            raise HermesApiError(f"locked: {name} is this agent's role skill and stays on")
    return {"method": method, "path": path, "query": query, "body": body, "home": home}
```

```python
    params: dict,
    *,
    config: HermesApiConfig,
    hermes_home: Path,
    transport: httpx.AsyncBaseTransport | None = None,
) -> dict:
    request = prepare_request(params, hermes_home=hermes_home)
    if not config.token:

    ) as client:

            response = await client.request(
                request["method"],
                request["path"],
                params=request["query"] or None,
                json=request["body"],
                headers={"X-Hermes-Session-Token": config.token},
            )

            raise HermesApiError(f"unavailable: {error}") from error

    if content_type == "application/json" or not response.content:
```

```python
MAX_AUDIO_BYTES = 5 * 1024 * 1024
TRANSCRIBE_TIMEOUT_SECONDS = 120.0
TRANSCRIPT_LIMIT = 200


async def transcribe_audio_via_hermes(
    data: bytes,
    mime_type: str,
    *,
    profile: str,
    config: HermesApiConfig,
    transport: httpx.AsyncBaseTransport | None = None,
) -> str:
    if len(data) > MAX_AUDIO_BYTES:
        raise HermesApiError(f"too_large: voice notes are limited to {MAX_AUDIO_BYTES // (1024 * 1024)} MB")
    if not config.token:
        raise HermesApiError("unavailable: the Hermes dashboard API is not configured on this host")
    mime = (mime_type or "audio/m4a").split(";")[0].strip() or "audio/m4a"
    payload = {"data_url": f"data:{mime};base64,{base64.b64encode(data).decode('ascii')}", "mime_type": mime}
    async with httpx.AsyncClient(base_url=config.base_url, transport=transport, timeout=TRANSCRIBE_TIMEOUT_SECONDS) as client:
        try:
            response = await client.post(
                "/api/audio/transcribe",
                params={"profile": normalize_profile(profile)},
                json=payload,
                headers={"X-Hermes-Session-Token": config.token},
            )
        except httpx.HTTPError as error:
            raise HermesApiError(f"unavailable: {error}") from error
    body = response.json() if response.content else {}
    if response.status_code != 200:
        raise HermesApiError(f"failed: {body.get('detail') or 'transcription failed'}")
    return str(body.get("transcript") or "").strip()


async def session_transcript(
    session_id: str,
    *,
    profile: str,
    config: HermesApiConfig,
    hermes_home: Path,
    transport: httpx.AsyncBaseTransport | None = None,
) -> dict:
    if not re.match(f"^{_SEG}$", session_id or ""):
        raise HermesApiError("invalid: bad session id")
    result = await call_hermes_api(
        {
            "method": "GET",
            "path": f"/api/sessions/{session_id}/messages",
            "query": {"profile": profile, "limit": str(TRANSCRIPT_LIMIT), "order": "latest"},
        },
        config=config,
        hermes_home=hermes_home,
        transport=transport,
    )
    if result["status"] != 200:
        raise HermesApiError(f"not_found: session {session_id} is not available")
    messages = []
    for item in (result.get("json") or {}).get("messages") or []:
        role = item.get("role")

            continue
        text = str(item.get("display_content") or item.get("content") or "").strip()
        if not text:
            continue
        messages.append({"id": str(item.get("id")), "role": role, "text": text, "timestamp": float(item.get("timestamp") or 0)})
    return {"sessionId": session_id, "messages": messages}
```

```python
    """The non-archived conversation the user activated most recently for this agent."""
    return db.scalar(
        _active_conversations(user_id, profile)

def open_session_conversation(db: Session, *, user_id: str, profile: str, session_id: str, title: str | None = None) -> Conversation:
    """Bind a fresh relay conversation to an existing Hermes session and make it current."""
    for old in db.scalars(
        _active_conversations(user_id, profile).where(Conversation.hermes_session_id == session_id)
    ):

        old.updated_at = utcnow()

        user_id=user_id,
        title=(title or NEW_CONVERSATION_TITLE),
        profile=profile,
        hermes_session_id=session_id,
        activated_at=utcnow(),
    )
    db.add(conversation)
    db.commit()
    db.refresh(conversation)
    return conversation
```

```swift
import Foundation

/// The user's Hermes agents (profiles) and which one the app is talking to.
@MainActor
@Observable
final class AgentStore {
    private static let selectionKey = "hermes.selectedAgent"

    private(set) var profiles: [HermesProfile] = []
    private(set) var selected: String
    var onSelectionChanged: (@MainActor (String) async -> Void)?

    private let api: HermesWorkspaceAPI
    private let defaults: UserDefaults

    init(api: HermesWorkspaceAPI, defaults: UserDefaults = .standard) {
        self.api = api
        self.defaults = defaults
        selected = defaults.string(forKey: Self.selectionKey) ?? RoleStyle.defaultProfileName
    }

    var ordered: [HermesProfile] {
        let rank = Dictionary(uniqueKeysWithValues: RoleStyle.preferredOrder.enumerated().map { ($1, $0) })
        return profiles.sorted { lhs, rhs in
            switch (rank[lhs.name], rank[rhs.name]) {
            case let (l?, r?): l < r
            case (_?, nil): true
            case (nil, _?): false
            default: lhs.name < rhs.name
            }
        }
    }

    var selectedStyle: RoleStyle { RoleStyle.forProfile(selected) }

    func refresh() async {
        guard let loaded = try? await api.profiles(), !loaded.isEmpty else { return }
        profiles = loaded
        if !loaded.contains(where: { $0.name == selected }) {
            select(RoleStyle.defaultProfileName)
        }
    }

    func select(_ name: String) {
        guard name != selected else { return }
        selected = name
        defaults.set(name, forKey: Self.selectionKey)
        let callback = onSelectionChanged
        Task { await callback?(name) }
    }

    #if DEBUG
    func setProfilesForTesting(_ names: [String]) {
        profiles = names.map { HermesProfile(name: $0, isDefault: $0 == RoleStyle.defaultProfileName, model: nil, provider: nil, description: nil) }
    }
    #endif
}
```

```swift
import Foundation

/// Chat history for one agent: its Hermes sessions, including Telegram and Mac app chats.
@MainActor
@Observable
final class ConversationsStore {
    struct Group: Identifiable, Equatable {
        let title: String
        let items: [HermesSessionSummary]
        var id: String { title }
    }

    private(set) var sessions: [HermesSessionSummary] = []
    private(set) var isLoading = false
    private(set) var isOffline = false
    var currentSessionID: String?

    /// Called after the current conversation changes, so chat can reload.
    var onConversationSwitched: (@MainActor () async -> Void)?

    private let api: HermesWorkspaceAPI

    init(api: HermesWorkspaceAPI) {
        self.api = api
    }

    func grouped(query: String = "", now: Date = Date(), calendar: Calendar = .current) -> [Group] {
        let needle = query.trimmingCharacters(in: .whitespaces)
        let matches = sessions.filter { session in
            needle.isEmpty
                || session.displayTitle.localizedCaseInsensitiveContains(needle)
                || (session.preview ?? "").localizedCaseInsensitiveContains(needle)
        }
        var today: [HermesSessionSummary] = []
        var week: [HermesSessionSummary] = []
        var earlier: [HermesSessionSummary] = []
        let weekAgo = now.addingTimeInterval(-7 * 24 * 3600)
        for session in matches {
            let date = session.lastActive ?? .distantPast
            if calendar.isDate(date, inSameDayAs: now) {
                today.append(session)
            } else if date > weekAgo {
                week.append(session)
            } else {
                earlier.append(session)
            }
        }
        return [Group(title: "Today", items: today), Group(title: "This week", items: week), Group(title: "Earlier", items: earlier)]
            .filter { !$0.items.isEmpty }
    }

    func refresh(profile: String) async {
        isLoading = true
        defer { isLoading = false }
        do {
            sessions = try await api.sessions(profile: profile)
            isOffline = false
        } catch {
            isOffline = true
        }
    }

    func reset() {
        sessions = []
        currentSessionID = nil
        isOffline = false
    }

    func startNew(profile: String) async throws {
        try await api.newConversation(profile: profile)
        currentSessionID = nil
        await onConversationSwitched?()
    }

    func open(_ session: HermesSessionSummary, profile: String) async throws {
        guard session.id != currentSessionID else { return }
        try await api.openSession(profile: profile, id: session.id)
        currentSessionID = session.id
        await onConversationSwitched?()
    }

    func rename(_ session: HermesSessionSummary, to title: String, profile: String) async throws {
        let trimmed = title.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { return }
        try await api.renameSession(id: session.id, title: trimmed, profile: profile)
        await refresh(profile: profile)
    }

    func archive(_ session: HermesSessionSummary, profile: String) async throws {
        try await api.archiveSession(id: session.id, profile: profile)
        await refresh(profile: profile)
        if session.id == currentSessionID {
            try await startNew(profile: profile)
        }
    }
}
```

```swift
import AVFoundation
import Foundation

/// Records a short voice note as AAC (.m4a) for transcription on the server.
@MainActor
@Observable
final class VoiceNoteRecorder {
    static let maxDuration: TimeInterval = 300

    private(set) var isRecording = false
    private(set) var elapsed: TimeInterval = 0

    private var recorder: AVAudioRecorder?
    private var timer: Timer?
    private var fileURL: URL?

    func start() async throws {
        guard await AVAudioApplication.requestRecordPermission() else {
            throw NSError(domain: "VoiceNote", code: 1, userInfo: [NSLocalizedDescriptionKey: "Microphone access is off. Turn it on in Settings."])
        }
        let session = AVAudioSession.sharedInstance()
        try session.setCategory(.playAndRecord, mode: .spokenAudio, options: [.defaultToSpeaker, .allowBluetoothHFP])
        try session.setActive(true)
        let url = FileManager.default.temporaryDirectory.appendingPathComponent("voice-\(UUID().uuidString).m4a")
        let settings: [String: Any] = [
            AVFormatIDKey: kAudioFormatMPEG4AAC,
            AVSampleRateKey: 16_000,
            AVNumberOfChannelsKey: 1,
            AVEncoderBitRateKey: 32_000,
        ]
        let recorder = try AVAudioRecorder(url: url, settings: settings)
        guard recorder.record(forDuration: Self.maxDuration) else {
            throw NSError(domain: "VoiceNote", code: 2, userInfo: [NSLocalizedDescriptionKey: "Couldn't start recording."])
        }
        self.recorder = recorder
        fileURL = url
        elapsed = 0
        isRecording = true
        timer = Timer.scheduledTimer(withTimeInterval: 0.5, repeats: true) { [weak self] _ in
            Task { @MainActor in self?.elapsed = self?.recorder?.currentTime ?? 0 }
        }
    }

    func stop() throws -> Data {
        timer?.invalidate()
        timer = nil
        recorder?.stop()
        recorder = nil
        isRecording = false
        try? AVAudioSession.sharedInstance().setActive(false, options: .notifyOthersOnDeactivation)
        guard let url = fileURL else { return Data() }
        defer { try? FileManager.default.removeItem(at: url) }
        return try Data(contentsOf: url)
    }
}
```
